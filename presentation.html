代码示例

@RestController
@RequestMapping("/api/goods")
public class GoodsController {
@Autowired
private GoodsService goodsService;
@GetMapping("/list")
public List<Goods> list() {
return goodsService.lambdaQuery()
.eq(Goods::getIsDelete, 0)
.orderByDesc(Goods::getCreateTime)
.list();
}
}
